== 1. 换算规则实现 ==
usd_cents = amount_cents × fx_rate，用 decimal.Decimal 做十进制精确乘法，
再 quantize(Decimal("1"), rounding=ROUND_HALF_UP)（.5 远离零进位），最后 int()。
全过程无 float，无二进制浮点误差。

== 6. 可重跑脚本（把以下内容存为 recompute.py 放在本工作目录，python recompute.py 即可复现）==
import csv, json
from decimal import Decimal, ROUND_HALF_UP

def usd_cents(amount_cents, fx_rate):
    return int((Decimal(amount_cents) * Decimal(fx_rate))
               .quantize(Decimal("1"), rounding=ROUND_HALF_UP))

rows = list(csv.DictReader(open("input/transactions.csv", encoding="utf-8-sig", newline="")))
accounts, monthly = {}, {}
for r in rows:
    c = usd_cents(r["amount_cents"], r["fx_rate"])
    a = accounts.setdefault(r["account"], {"count": 0, "usd_cents": 0, "best": (-10**18, "")})
    a["count"] += 1
    a["usd_cents"] += c
    if c > a["best"][0] or (c == a["best"][0] and r["txn_id"] < a["best"][1]):
        a["best"] = (c, r["txn_id"])
    m = monthly.setdefault(r["date"][:7], {"count": 0, "usd_cents": 0})
    m["count"] += 1
    m["usd_cents"] += c

out = {
    "accounts": {k: {"count": v["count"], "usd_cents": v["usd_cents"], "largest_txn": v["best"][1]}
                 for k, v in accounts.items()},
    "monthly": monthly,
    "grand_total_cents": sum(v["usd_cents"] for v in accounts.values()),
}
with open("aggregate.json", "w", encoding="utf-8", newline="\n") as f:
    json.dump(out, f, ensure_ascii=False, indent=2)
    f.write("\n")
assert out["grand_total_cents"] == sum(v["usd_cents"] for v in monthly.values())
print("OK", out["grand_total_cents"])
